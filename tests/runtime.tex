\directlua{
  local path = assert(kpse.find_file("azhyphen.lua", "tex"))
  local first = assert(loadfile(path))()
  local second = assert(loadfile(path))()
  assert(first == second, "module caching failed")
  local id = first.activate()
  assert(id == second.activate() and id == tex.language, "global activation failed")
  local object = lang.new(id)
  for _, pair in ipairs({{"ç", "Ç"}, {"ə", "Ə"}, {"ğ", "Ğ"},
      {"i", "İ"}, {"ı", "I"}, {"ö", "Ö"}, {"ş", "Ş"}, {"ü", "Ü"}}) do
    local code = utf8.codepoint(pair[1])
    assert(lang.gethjcode(object, code) == code)
    assert(lang.gethjcode(object, utf8.codepoint(pair[2])) == code)
  end
}
