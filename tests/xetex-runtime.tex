\def\checkcase#1#2{%
  \ifnum\lccode`#2=`#1\relax\else
    \PackageError{azhyphen-test}{Incorrect lowercase mapping}{}%
  \fi
  \ifnum\uccode`#1=`#2\relax\else
    \PackageError{azhyphen-test}{Incorrect uppercase mapping}{}%
  \fi}
